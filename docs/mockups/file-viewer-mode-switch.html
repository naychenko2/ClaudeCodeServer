import { useEffect, useMemo, useState } from 'react';
import type { Project, Session } from '../types';
import { api } from '../lib/api';
import { useOnline } from '../hooks/useOnline';

export function SessionList({ project, activeSession, onSelect }: Props) {
  const [sessions, setSessions] = useState<Session[]>([]);

  useEffect(() => {
    const init = async () => {
      const list = await api.sessions.list(project.id);
      setSessions(list);
    };
    init();
  }, [project.id]);

  return (
    <div>
      {sessions.map(session => (
        <ChatCard
          key={session.id}
          session={session}
          active={session.id === activeSession?.id}
          onClick={() => onSelect(session)}
        />
      ))}
    </div>
  );
}

Мобильная раскладка (≤600px) Только полный режим, переключатель скрыт

useIsMobile (MOBILE_MAX=600): На мобильных устройствах экран слишком узкий для сплита, поэтому показываем только полный режим. Переключатель скрыт через CSS (desktop-only-controls).
